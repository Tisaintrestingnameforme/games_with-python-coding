🐍 El Python detrás del juego

Cuando cargue Python, será Python quien calcule hacia dónde se mueve el personaje y si atrapó una estrella.

import math

def mover(x, y, dx, dy, estrella_x, estrella_y):
    nx = max(18, min(702, x + dx))
    ny = max(18, min(392, y + dy))
    atrapada = math.hypot(nx - estrella_x,
                          ny - estrella_y) < 35
    return f"{nx},{ny},{int(atrapada)}"
